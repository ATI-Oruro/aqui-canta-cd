# Iniciar Base de Datos (Podman)
podman compose up -d
# (o podman-compose up -d)
podman start acb_mysql acb_phpmyadmin


# Poblar datos (solo primera vez o reset)
php artisan migrate --seed

# Servidor Laravel
php artisan serve

# Servidor Websocket (Reverb)
php artisan reverb:start
php artisan reverb:start --host=0.0.0.0 --port=6080

# Frontend (Vite)
npm run dev
